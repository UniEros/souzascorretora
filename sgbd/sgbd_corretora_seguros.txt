2. SELECT - CONSULTAR DADOS
============================================================

-- 2.1. Consultar todos os funcionários
SELECT * FROM funcionario;

-- 2.2. Consultar segurados
SELECT id_cliente, nome, cpf, ocupacao, renda_mensal
FROM segurado;

-- 2.3. Segurados com renda superior a R$ 3.000
SELECT nome, ocupacao, renda_mensal
FROM segurado
WHERE renda_mensal > 3000.00
ORDER BY renda_mensal DESC;

-- 2.4. Seguros, segurados e funcionários responsáveis
SELECT
    s.id_seguro,
    c.nome AS segurado,
    f.nome AS funcionario,
    s.tipo_seguro,
    s.vigencia_inicial,
    s.vigencia_final
FROM seguro s
INNER JOIN segurado c
    ON s.id_cliente = c.id_cliente
INNER JOIN funcionario f
    ON s.id_funcionario = f.id_funcionario;

-- 2.5. Veículos e seus seguros
SELECT
    v.marca,
    v.modelo,
    v.placa,
    v.ano_modelo,
    s.tipo_seguro,
    s.vigencia_final
FROM veiculo v
INNER JOIN seguro s
    ON v.id_seguro = s.id_seguro;

-- 2.6. Contatos de cada segurado
SELECT
    s.nome,
    c.email,
    c.telefone_celular,
    c.telefone_fixo
FROM segurado s
LEFT JOIN contato c
    ON s.id_cliente = c.id_cliente;

-- 2.7. Seguros vigentes na data atual
SELECT *
FROM seguro
WHERE CURDATE() BETWEEN vigencia_inicial AND vigencia_final;

-- 2.8. Quantidade de seguros por tipo
SELECT
    tipo_seguro,
    COUNT(*) AS quantidade
FROM seguro
GROUP BY tipo_seguro;

-- 2.9. Imóveis com área construída superior a 100 m²
SELECT
    id_imovel,
    logradouro,
    cidade,
    uf,
    area_construida
FROM imovel
WHERE area_construida > 100;

-- 2.10. Listar as tabelas do banco
SHOW TABLES;

3. ALTER TABLE - ALTERAR A ESTRUTURA
============================================================

-- 3.1. Adicionar e-mail ao funcionário
ALTER TABLE funcionario
ADD COLUMN email VARCHAR(100);

-- 3.2. Aumentar o tamanho do nome do segurado
ALTER TABLE segurado
MODIFY COLUMN nome VARCHAR(150) NOT NULL;

-- 3.3. Adicionar valor do prêmio ao seguro
ALTER TABLE seguro
ADD COLUMN valor_premio DECIMAL(10,2);

-- 3.4. Adicionar observações ao funcionário
ALTER TABLE funcionario
ADD COLUMN observacoes TEXT;

-- 3.5. Remover a coluna de observações criada no exemplo 3.4
-- ATENÇÃO: este comando apaga a coluna e os dados nela armazenados.
ALTER TABLE funcionario
DROP COLUMN observacoes;


============================================================
4. DELETE - EXCLUIR REGISTROS
============================================================

-- ATENÇÃO:
-- Execute DELETE somente em dados de teste ou após fazer backup.
-- A cláusula WHERE é importante para não excluir todos os registros.

-- 4.1. Excluir um contato específico
DELETE FROM contato
WHERE id_contato = 1;

-- 4.2. Excluir um veículo pela placa
-- Se houver condutor associado, exclua o condutor primeiro.
DELETE FROM veiculo
WHERE placa = 'ABC1D23';

-- 4.3. Excluir todos os contatos de um segurado
DELETE FROM contato
WHERE id_cliente = 1;

-- 4.4. Excluir o seguro de ID 1 e seus registros dependentes
-- Execute este bloco apenas se quiser apagar os dados de teste associados.
START TRANSACTION;

DELETE FROM condutor_principal
WHERE id_veiculo IN (
    SELECT id_veiculo
    FROM veiculo
    WHERE id_seguro = 1
);

DELETE FROM questionario_imovel
WHERE id_imovel IN (
    SELECT id_imovel
    FROM imovel
    WHERE id_seguro = 1
);

DELETE FROM veiculo WHERE id_seguro = 1;
DELETE FROM imovel WHERE id_seguro = 1;
DELETE FROM viagem WHERE id_seguro = 1;
DELETE FROM bicicleta WHERE id_seguro = 1;
DELETE FROM cobertura WHERE id_seguro = 1;

DELETE FROM seguro WHERE id_seguro = 1;

COMMIT;

-- 4.5. Excluir um segurado e seus dados relacionados
-- Pressupõe que os seguros desse cliente já foram removidos.
START TRANSACTION;

DELETE FROM contato
WHERE id_cliente = 1;

DELETE FROM perfil
WHERE id_cliente = 1;

DELETE FROM segurado
WHERE id_cliente = 1;

COMMIT;

-- 4.6. Excluir um funcionário
-- Só funciona se não houver seguros vinculados a ele.
DELETE FROM funcionario
WHERE id_funcionario = 1;


============================================================
5. RESUMO DOS COMANDOS
============================================================

INSERT INTO       - Insere registros.
SELECT            - Consulta dados.
WHERE             - Filtra registros.
ORDER BY          - Ordena resultados.
INNER JOIN        - Consulta registros relacionados em tabelas.
LEFT JOIN         - Mantém todos os registros da tabela à esquerda.
GROUP BY          - Agrupa resultados.
COUNT()           - Conta registros.
ALTER TABLE       - Altera a estrutura de uma tabela.
ADD COLUMN        - Adiciona uma coluna.
MODIFY COLUMN     - Modifica uma coluna.
DROP COLUMN       - Remove uma coluna.
DELETE FROM       - Exclui registros.
START TRANSACTION - Inicia uma transação.
COMMIT            - Confirma as operações da transação.


============================================================
OBSERVAÇÕES IMPORTANTES
============================================================

1. Os dados usados nos INSERTs são fictícios.
2. Os INSERTs informam IDs manualmente para facilitar os exemplos.
   Execute-os em um banco vazio ou ajuste os IDs conforme os dados existentes.
3. Os exemplos associam um mesmo seguro a veículo, imóvel, viagem e bicicleta
   apenas para demonstrar os relacionamentos. Em um sistema real, use vínculos
   coerentes com a modalidade de cada seguro.
4. Os ALTER TABLE de adição de colunas devem ser executados apenas uma vez.
5. Os DELETEs removem dados. Teste em uma cópia do banco e faça backup antes.
6. As chaves estrangeiras impedem excluir registros que ainda tenham registros
   dependentes, a menos que a operação seja tratada primeiro.
7. O campo senha deve guardar um hash seguro, e não uma senha em texto puro.
